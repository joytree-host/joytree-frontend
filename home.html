# Deploy via API — GitHub or upload curl -X POST https://joytree.site/api/deploy \ -H "Content-Type: application/json" \ -d '{ "name": "my-react-app", "subdomain": "myapp", "repoUrl": "https://github.com/user/repo", "branch": "main", "buildCmd": "npm run build" }' # ✓ Live at myapp.joytree.site in ~30s
// Managed database — created instantly { "type": "redis", "name": "session-cache", "memoryMb": 256, "status": "running", "host": "localhost", "port": 6380 } // Also supports: MongoDB, MySQL, PostgreSQL
# Plain-English API prompt "Create a REST API that returns 10 trivia questions with answers, categorized by difficulty. Include a /random endpoint." # JoyTree generates + deploys instantly: # GET /api/live/flow_1780257588074 # GET /api/live/flow_1780257588074/random
# Instant rollback — no rebuild needed curl -X POST https://joytree.site/api/projects/myapp/rollback \ -H "Authorization: Bearer <token>" \ -d '{ "deployId": "prev-deploy-abc" }' # Response: { ok: true, rolledBackTo: "prev-deploy-abc" } # Zero downtime. Zero re-cloning.
